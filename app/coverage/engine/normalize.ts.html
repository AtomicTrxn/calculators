/** Optional per-line age bound: null/''/missing means "no bound". */
function optionalAge(v: unknown): number | null {
  if (v == null || v === '') return null
  const n = Math.round(Number(v))
  return Number.isFinite(n) ? clamp(n, 0, 120) : null
}
